cmake_minimum_required(VERSION 3.16)
project(p25scanner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(PkgConfig REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
pkg_check_modules(SOAPY REQUIRED SoapySDR)
pkg_check_modules(FFTW REQUIRED fftw3f)
add_executable(p25scanner main.cpp)
add_executable(p25scanner_gui gui.cpp)
foreach(target p25scanner p25scanner_gui)
  target_include_directories(${target} PRIVATE ${SOAPY_INCLUDE_DIRS} ${FFTW_INCLUDE_DIRS})
  target_link_directories(${target} PRIVATE ${SOAPY_LIBRARY_DIRS} ${FFTW_LIBRARY_DIRS})
  target_link_libraries(${target} PRIVATE ${SOAPY_LIBRARIES} ${FFTW_LIBRARIES})
  target_compile_options(${target} PRIVATE -O3 -Wall -Wextra)
endforeach()
target_link_libraries(p25scanner_gui PRIVATE Qt5::Widgets)
