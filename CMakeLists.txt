cmake_minimum_required(VERSION 3.16)
project(p25scanner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SOAPY REQUIRED SoapySDR)
pkg_check_modules(FFTW REQUIRED fftw3f)
add_executable(p25scanner main.cpp)
target_include_directories(p25scanner PRIVATE ${SOAPY_INCLUDE_DIRS} ${FFTW_INCLUDE_DIRS})
target_link_directories(p25scanner PRIVATE ${SOAPY_LIBRARY_DIRS} ${FFTW_LIBRARY_DIRS})
target_link_libraries(p25scanner PRIVATE ${SOAPY_LIBRARIES} ${FFTW_LIBRARIES})
target_compile_options(p25scanner PRIVATE -O3 -Wall -Wextra)
